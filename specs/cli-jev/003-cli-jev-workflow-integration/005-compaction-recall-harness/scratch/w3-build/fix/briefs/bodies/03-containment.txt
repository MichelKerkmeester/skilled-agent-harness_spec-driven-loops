function isInsideByIdentity(target, candidate) {
  let current = candidate;
  while (true) {
    if (existsSync(current)) {
      const stats = statSync(current);
      if (stats.dev === target.dev && stats.ino === target.ino) {
        return true;
      }
    }
    const parent = dirname(current);
    if (parent === current) {
      return false;
    }
    current = parent;
  }
}
