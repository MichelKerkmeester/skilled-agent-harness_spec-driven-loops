## probe plugin source (placed as .opencode/plugins/zz-event-probe.js for the two runs, then deleted)
import { appendFileSync } from 'node:fs';
import { dualPlugin } from './lib/v2-bridge.js';
const OUT = '<scratch>/events.log';
async function probe(ctx) {
  appendFileSync(OUT, 'factory ' + Date.now() + '\n');
  return {
    async event(input) { appendFileSync(OUT, 'event ' + Date.now() + ' ' + (input?.event?.type) + ' ' + JSON.stringify(input?.event?.properties ?? {}).slice(0, 160) + '\n'); },
    async 'tool.execute.before'(input, output) { appendFileSync(OUT, 'before ' + input.tool + ' ' + JSON.stringify(output.args).slice(0, 200) + '\n'); },
    async 'experimental.chat.system.transform'(input) { appendFileSync(OUT, 'transform ' + JSON.stringify(input).slice(0, 200) + '\n'); },
  };
}
export default dualPlugin('zz-event-probe', probe);
