import { appendFileSync } from "node:fs";

const LOG = "/Users/michelkerkmeester/MEGA/Development/Code_Environment/Public/.worktrees/143-opencode-dual-compat-build/specs/cli-external-orchestration/082-opencode-v1-v2-dual-compat/001-unknowns-and-baseline/scratch/q7-marker.log";

function write(obj) {
  try {
    appendFileSync(LOG, JSON.stringify(obj) + "\n");
  } catch {}
}

const JSON_SCHEMA_INPUT = {
  $schema: "https://json-schema.org/draft/2020-12/schema",
  type: "object",
  properties: { text: { type: "string" } },
  required: ["text"],
};

export default {
  id: "q7-jsonschema-tool",
  async setup(ctx) {
    write({ t: Date.now(), evt: "setup-fired", pid: process.pid });
    try {
      ctx.tool.transform((editor) => {
        editor.add({
          name: "q7_echo",
          description: "Probe tool whose input is a z.toJSONSchema output (plain JSON Schema).",
          input: JSON_SCHEMA_INPUT,
          execute: async (input) => ({ echoed: input?.text ?? null }),
        });
        return editor;
      });
      write({ t: Date.now(), evt: "transform-registered" });
    } catch (e) {
      write({ t: Date.now(), evt: "transform-register-threw", message: String(e?.message ?? e) });
    }
    try {
      const tools = await ctx.tool.list();
      write({
        t: Date.now(),
        evt: "tool-list-ok",
        count: tools.length,
        names: tools.map((t) => t?.name).slice(0, 40),
        hasQ7: tools.some((t) => t?.name === "q7_echo"),
      });
    } catch (e) {
      write({ t: Date.now(), evt: "tool-list-threw", message: String(e?.message ?? e) });
    }
    return () => {};
  },
};
