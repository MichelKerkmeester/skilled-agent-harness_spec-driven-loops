import { appendFileSync } from "node:fs";

const LOG = "/Users/michelkerkmeester/MEGA/Development/Code_Environment/Public/.worktrees/143-opencode-dual-compat-build/specs/cli-external-orchestration/082-opencode-v1-v2-dual-compat/001-unknowns-and-baseline/scratch/q5-q6-events.log";

function write(obj) {
  try {
    appendFileSync(LOG, JSON.stringify(obj) + "\n");
  } catch {}
}

export default {
  id: "q5-q6-event-logger",
  setup(ctx) {
    write({ t: Date.now(), evt: "setup-fired", pid: process.pid });
    const dispose = ctx.event.subscribe((event) => {
      let raw = "";
      try {
        raw = JSON.stringify(event);
      } catch {
        raw = "<unserializable>";
      }
      write({
        t: Date.now(),
        evt: event?.type ?? "no-type-field",
        keys: event && typeof event === "object" ? Object.keys(event) : typeof event,
        dataKeys: event?.data && typeof event.data === "object" ? Object.keys(event.data) : null,
        raw: raw.slice(0, 300),
      });
    });
    write({ t: Date.now(), evt: "subscribe-returned", kind: typeof dispose });
    return () => {};
  },
};
