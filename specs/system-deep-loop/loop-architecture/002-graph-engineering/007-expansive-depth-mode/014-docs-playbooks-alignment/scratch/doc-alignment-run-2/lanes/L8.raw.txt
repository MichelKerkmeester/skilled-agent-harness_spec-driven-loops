function normalizeDepthMode(raw) {
  if (raw === undefined || raw === false || raw === null || raw === '' || raw === 'default') return null;
  if (raw !== 'expansive') throw inputError('depthMode must be one of: default, expansive');
  return raw;
}

// Open depth-1 nodes are dealt round-robin in node-id order, so the same seed
// always gives each lineage the same share and the merge guard can check it.
function partitionFrontierNodes(baseArtifactDir, labels) {
  const { readNodeStore } = require('./frontier-admit.cjs');
  const nodes = [...readNodeStore(path.join(baseArtifactDir, 'frontier-nodes.jsonl')).values()]
    .filter((node) => node.depth === 1 && (node.status === 'open' || node.status === 'active'))
    .sort((left, right) => left.nodeId.localeCompare(right.nodeId));
  const partition = Object.fromEntries(labels.map((label) => [label, []]));
  nodes.forEach((node, index) => { partition[labels[index % labels.length]].push(node); });
  return partition;
}

// A lineage starts from its own store and inbox. They are written only when the
// lineage has no store yet, so a resumed lineage keeps the nodes it already closed.
// Review carries no inbox: its nodes reach the leaf through the FRONTIER block, so
// a seeded inbox would only hand the finding pipeline a file nothing reads.
function writeLineageFrontierStore(lineageDir, nodes, options = {}) {
  const storePath = path.join(lineageDir, 'frontier-nodes.jsonl');
  if (fs.existsSync(storePath) || nodes.length === 0) return false;
  fs.mkdirSync(lineageDir, { recursive: true });
  fs.writeFileSync(storePath, nodes.map((node) => `${JSON.stringify(node)}\n`).join(''), 'utf8');
  if (options.inbox === false) return true;
  const inboxRows = nodes.map((node) => JSON.stringify({ id: node.nodeId, text: node.question, source: 'frontier', origin: 'frontier', injectedAtIteration: 0 }));
  fs.appendFileSync(path.join(lineageDir, 'inbox.jsonl'), `${inboxRows.join('\n')}\n`, 'utf8');
  return true;
}

### Expansive depth mode (--depth-mode=expansive)

#### Description

Opt-in --depth-mode=expansive on deep-review seeds depth-one sub-target nodes from the review target, keeps node status in review/frontier-nodes.jsonl, and holds a review stop while nodes stay open.

#### How It Works

Node status lives in `review/frontier-nodes.jsonl`, open nodes count as `searchDebt`, and the divergent pivot waits for a terminal frontier. The live pilot runs are still open, so no pilot result is recorded here.

### Completed-run extension (--extend)

#### Description

A completed deep-review run extends in place by N iterations under --extend=N, keeps its session identifier and generation, and keeps the prior review report as a snapshot before re-synthesis.

#### How It Works

The run continues under lineage mode `completed-continue`, keeps the prior review report as a snapshot before re-synthesis and takes `--extend-minutes=M` on the duration budget. Extension is refused with fan-out.

### Review minIterations floor

#### Description

A review run with a configured minIterations floor returns CONTINUE with minIterationsNotReached until its iteration count reaches the floor, so a review never stops before the floor.

#### How It Works

`convergence.cjs` returns `CONTINUE` with `minIterationsNotReached` below the floor, and the command's fixed stop order checks the floor before the gates. The shipped default is 2.

function lastClaimAdjudicationPassed(stateRows) {
  let lastMatch = null;
  for (const row of stateRows) {
    if (isPlainObject(row) && row.type === 'event' && row.event === 'claim_adjudication') {
      lastMatch = row;
    }
  }
  return lastMatch !== null && typeof lastMatch.passed === 'boolean' ? lastMatch.passed : null;
}

function reviewFallbackCoveredBugClasses(iterationRecords) {
  const evidenceMethods = new Set([
    'direct_read',
    'exact_grep',
    'semantic_search',
    'producer_consumer_trace',
    'negative_test_inspection',
  ]);
  const covered = new Set();
  for (const record of iterationRecords) {
    if (!Array.isArray(record.searchLedger)) continue;
    for (const row of record.searchLedger) {
      if (!isPlainObject(row) || typeof row.bugClass !== 'string') continue;
      if (!Array.isArray(row.evidenceRefs) || row.evidenceRefs.length === 0) continue;
      const actions = Array.isArray(row.searchActions) ? row.searchActions : [];
      if (!actions.some((action) => isPlainObject(action) && evidenceMethods.has(action.method))) {
        continue;
      }
      covered.add(row.bugClass);
    }
  }
  return [...covered].sort();
}

function graphHotspotSaturation(graph) {
  if (!isPlainObject(graph) || graph.failed === true) return null;
  const trace = Array.isArray(graph.trace) ? graph.trace : [];
  const entry = trace.find(
    (candidate) => isPlainObject(candidate) && candidate.signal === 'hotspotSaturation',
  );
