import re, sys
for path in sys.argv[1:]:
    lines = open(path).read().split("\n")
    start = lines.index("      step_acquire_lock:")
    out = lines[: start + 1]
    i = start + 1
    counts = {"halt": 0, "lost": 0, "counters": 0, "cancel": 0}
    while i < len(lines):
        line = lines[i]
        stripped = line.strip()
        indent = line[: len(line) - len(line.lstrip())]
        if "fail closed: lost the packet lock" in line and "step_release_lock" not in line:
            new = line.replace(
                "This run no longer holds the packet and must stop",
                "This run no longer holds the packet, so run step_release_lock and stop",
            )
            if new == line:
                raise SystemExit(f"{path}:{i + 1}: lost-lock line has an unexpected wording")
            line = new
            counts["lost"] += 1
        if stripped.startswith("on_missing_counters:") and "step_release_lock" not in line:
            if not line.endswith('"'):
                raise SystemExit(f"{path}:{i + 1}: on_missing_counters line does not end with a quote")
            line = line[:-1] + ' Run step_release_lock before the halt returns control."'
            counts["counters"] += 1
        if 'status: "CANCELLED"' in line and "release_lock: step_release_lock" not in line:
            line = line.replace('status: "CANCELLED"', 'status: "CANCELLED", release_lock: step_release_lock')
            counts["cancel"] += 1
        out.append(line)
        if stripped == "halt: true":
            nxt = lines[i + 1] if i + 1 < len(lines) else ""
            if nxt.strip() != "release_lock: step_release_lock":
                out.append(indent + "release_lock: step_release_lock")
                counts["halt"] += 1
        i += 1
    open(path, "w").write("\n".join(out))
    print(path, counts)
