        writeMarkerSleepingStubBinary(binDir, 'opencode', 0, markerPath);

          executors: [{ label: 'only-seat', kind: 'cli-opencode', model: 'opencode-go/glm-5.1', iterations: 1 }],
          concurrency: 1,
        });

          '--spec-folder', specFolder, '--loop-type', loopType,
