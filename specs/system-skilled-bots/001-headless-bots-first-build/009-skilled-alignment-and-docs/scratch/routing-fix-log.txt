)
s1() {
  printf '%s' "$(node -e 'process.stdout.write(JSON.stringify({prompt:process.argv[1]}))' "$1")" \
    | node .skilled/bin/skill-advisor.cjs advisor_recommend --json - --format json 2>/dev/null \
    | node -e 'let s="";process.stdin.on("data",d=>s+=d).on("end",()=>{try{const r=JSON.parse(s).data.recommendations||[];const top=r[0];const i=r.findIndex(x=>x.skillId==="system-skilled-bots");const j=r.findIndex(x=>x.skillId==="system-skilled-bots"||/^command-bots-/.test(x.skillId));process.stdout.write((top?top.skillId+"@"+(top.confidence??top.score??"")+(top.command?"["+top.command+"]":""):"none")+" bots_rank="+(j<0?"-":j+1))}catch(e){process.stdout.write("ERR "+e.message)}})'
}
s2() {
  node .skilled/bin/compiled-route.cjs --hub system-skilled-bots --prompt "$1" 2>/dev/null \
    | node -e 'let s="";process.stdin.on("data",d=>s+=d).on("end",()=>{try{const r=JSON.parse(s);const m=(r.targets||[]).map(t=>t.workflowMode||(t.destinationId&&t.destinationId.workflowMode)||JSON.stringify(t)).join("+");process.stdout.write(r.action+(m?":"+m:"")+(r.selectionKind?"("+r.selectionKind+")":"")+" hash="+String(r.effectivePolicyHash||r.servingAuthority||"").slice(0,12))}catch(e){process.stdout.write("ERR "+e.message)}})'
}
inpass=0; infail=0; oodfail=0; adjfail=0
for row in "${IN[@]}"; do
  exp="${row%%|*}"; p="${row#*|}"
  a=$(s1 "$p"); b=$(s2 "$p")
  ok1=no; case "$a" in system-skilled-bots@*|command-bots-*) ok1=yes;; esac
  ok2=no; case "$b" in "route:$exp("*) ok2=yes;; esac
  if [ $ok1 = yes ] && [ $ok2 = yes ]; then inpass=$((inpass+1)); else infail=$((infail+1)); fi
  echo "IN  exp=$exp | s1 top=$a ok=$ok1 | s2 $b ok=$ok2 | $p"
done
for q in "${OOD[@]}"; do
  tag=OOD; p="$q"; case "$q" in ADJ:*) tag=ADJ; p="${q#ADJ:}";; esac
  a=$(s1 "$p"); b=$(s2 "$p")
  bad1=no; case "$a" in system-skilled-bots@*|command-bots-*|command-doctor-bots@*) bad1=yes;; esac
  bad2=no; case "$b" in route*) bad2=yes;; esac
  if [ $tag = OOD ] && { [ $bad1 = yes ] || [ $bad2 = yes ]; }; then oodfail=$((oodfail+1)); fi
  if [ $tag = ADJ ] && { [ $bad1 = yes ] || [ $bad2 = yes ]; }; then adjfail=$((adjfail+1)); fi
  echo "$tag s1 top=$a bots-first=$bad1 | s2 $b routed=$bad2 | $p"
done
for t in "/bots:author weekly-digest --repo ~/code/acme" "/bots:run answer item-42 yes" "/bots:schedule pause nightly-cleanup" "/bots:status inbox --json"; do
  py=$(python3 .skilled/skills/system-skill-advisor/runtime/scripts/skill_advisor.py "$t" 2>/dev/null | python3 -c 'import json,sys; d=json.load(sys.stdin); print([(x["skill"],x.get("confidence")) for x in d[:3]])')
  a=$(s1 "$t")
  echo "TYPED py=$py | s1 top=$a | $t"
done
echo "SUMMARY in-pass=$inpass in-fail=$infail ood-fail=$oodfail adj-fail=$adjfail"

# For each named-bot probe, list every contiguous n-gram (after the matcher's lower-case and
# hyphen-to-space fold) that does not contain the bot's name, and the reason each one cannot
# serve as a stage-two keyword. The matcher is a plain substring test, so a keyword must be one
# of these n-grams to fire on the probe.

probes = {
    "Run the release-notes bot now against this repository": ("run", {"release", "notes"}),
    "Pause the triage bot until Monday": ("schedule", {"triage"}),
    "launch the triage bot": ("run", {"triage"}),

       "start the server","launch the app","pause the video","drain the queue","remove the cron job","show me which packages are installed",
       "read my inbox","my launchd agent is broken","pause the build until Monday","run the tests until they pass","pause the launchd agent",
       "launch the browser","write a system prompt for a support triage bot","list my chatbot runs","the robot needs you"]
settled = {"bot now": "excluded in 008 Key Decisions (captured doctor prompts in the advisor slice)", "hand": "settled drop", "question": "settled drop",
           "open": "settled drop", "launchd schedule": "settled drop"}
baseline_tokens = set("answer author bot bots create cron defined definition digest disable drain edit enable enqueue file four headless hub inbox installed json kick last launchd md mode modes needs new nightly off parent pause queue registry request resume routes run runs schedule skilled status system template through workflowmode write".split())
stop = set("a about able actually agent all also an and any are as at be been being but by can could did do does even for from get give go going had has have he help her him how i if in into is it its just let like may me might more most must my need no not now of on only or other our please really she should skill skills so some tell that the them then these they thing things this those to tool try us used using very want was way we were what when where which who why will with would you your".split())

for probe, (mode, names) in probes.items():
    words = fold(probe).split()
    print(f"== {probe!r} (expected {mode}; name tokens {sorted(names)})")
    for n in range(1, 5):
        for i in range(len(words) - n + 1):
            g = words[i:i+n]
            if names & set(g):
                continue
            gram = " ".join(g)
            content = [w for w in g if w not in stop and len(w) > 1]
            reasons = []
            if not content:
                reasons.append("stop words only: matches almost any prompt")
            hits = [o for o in ood if gram in fold(o)]
            if hits:
                reasons.append("matches OOD " + json.dumps(hits[:2]))
            if gram in settled:
                reasons.append(settled[gram])
            if "bot" not in g and not hits and content:
                reasons.append("no bot token: nothing ties it to a bot request")
            new = [w for w in content if w not in baseline_tokens]
            if new and "bot" in g:
                reasons.append(f"adds stage-one token(s) {new}; trailing context shaped to this probe")
            if g == ["bot"]:
                reasons.append("names no mode: every bot request in every mode, plus robot/chatbot")
            print(f"   {gram!r:32} -> " + ("; ".join(reasons) if reasons else "NO REASON FOUND"))
